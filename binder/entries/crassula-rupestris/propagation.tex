\CompanionHeader{PROPAGATION}{Crassula rupestris}{Crassula rupestris}{Stem cutting first; leaf alternate. Proposed home sequence based on RHS and general extension guidance.}
\Context{Separate callusing, rooting and establishment. The nursery label does not guarantee a rooting deadline or confirm a cultivar. Choose healthy material and record the result.}
\begin{minipage}[t]{.625\linewidth}\raggedright\vspace{0pt}
% step-claims: crassula-rupestris#prepare
\Step{1}{Prepare}{Use clean sharp snips, a labeled draining pot, a dry tray and fresh cactus rooting mix. Work with a healthy plant. \textbf{[SU-ISU]}}
% step-claims: crassula-rupestris#select
\Step{2}{Select}{Choose a healthy stem that can be removed without dismantling the plant. If there is too little stem, postpone; leaf cuttings are an alternate. \textbf{[CR-RHS, SU-ISU]}}
% step-claims: crassula-rupestris#cut
\Step{3}{Cut / clear}{General ISU stem method: 3-6 in (7.6-15.2 cm), where the plant permits. Remove lower leaves; retain healthy upper foliage. \textbf{[SU-ISU]}}
% step-claims: crassula-rupestris#callus
\Step{4}{Callus}{Rest the cut stem on a dry tray for several days. Continue only when the cut surface is dry; this is a wound stage, not rooting. \textbf{[SU-ISU]}}
% step-claims: crassula-rupestris#set
\Step{5}{Set / label}{Place the callused base into freely draining rooting mix; keep foliage above it. Label parent, method and start date. \textbf{[SU-ISU]}}
% step-claims: crassula-rupestris#root
\Step{6}{Root / observe}{Provide bright indirect light initially. Keep rooting medium only slightly moist; avoid saturation and standing water. Do not use an adult watering calendar. \textbf{[SU-MSU]}}
% step-claims: crassula-rupestris#readiness
\Step{7}{Check readiness}{Wait for roots and continuing healthy growth. Gentle resistance can indicate anchoring; avoid repeated pulling. No fixed transplant date is promised. \textbf{[SU-MSU, SU-ISU]}}
% step-claims: crassula-rupestris#transfer
\Step{8}{Pot / acclimate}{Once rooted, move to a draining container; gradually adjust to established-plant light and watering. Record outcomes and delays. \textbf{[SU-MSU, SU-ISU]}}
\end{minipage}\hfill\begin{minipage}[t]{.345\linewidth}\raggedright\vspace{0pt}
\Note{Method scope}{RHS supports stem or leaf cuttings in spring or early summer; no species-specific rooting interval is asserted. [CR-RHS]}
\Note{Leaf alternate}{Use an entire healthy leaf with its stem attachment intact. Lay on slightly damp draining mix, attachment at the surface. Observe for roots and a new plant; partial leaves can fail. [SU-ISU]}
\Note{Check moisture}{Keep an unrooted cutting out of standing water. Persistent wetness warrants a drainage check, not another drink. [SU-MSU]}
\Note{Check light}{Start gently, then acclimate. Stretching suggests checking exposure; scorched tissue suggests checking sudden sun changes. [SU-MSU]}
\Note{Check progress}{No fixed root-initiation or transplant interval. Record firm tissue, roots and sustained new growth separately. [SU-MSU, SU-ISU]}
\end{minipage}\par\vspace{4pt}
